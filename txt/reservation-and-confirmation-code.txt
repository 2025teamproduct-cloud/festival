//  /api/reservations.js
import { cert, getApps, initializeApp } from 'firebase-admin/app';
import { getFirestore } from 'firebase-admin/firestore';
import { randomUUID } from 'node:crypto';


// ========================================
// 基本設定
// ========================================

// 1枠の定員（人数）
const SLOT_CAPACITY = 20;

// 1回の予約の最大人数（代表者＋同行者）
const MAX_PARTY_SIZE = 4;

// 枠の間隔（分）
const SLOT_INTERVAL_MINUTES = 30;


// ========================================
// 学園祭日程
// ========================================

const FESTIVAL_DATES = {
  day1: '2026-11-21',
  day2: '2026-11-22',
};


// ========================================
// 日ごとの営業時間（両日 11:00〜16:00）
// ========================================

const DAY_SETTINGS = {
  day1: {
    startTimeMinutes: 11 * 60,
    endTimeMinutes: 16 * 60,
  },

  day2: {
    startTimeMinutes: 11 * 60,
    endTimeMinutes: 16 * 60,
  },
};


// ========================================
// 指定日の時間枠を作成
// ========================================

function createSlots(day) {
  const setting = DAY_SETTINGS[day];

  if (!setting) {
    return [];
  }

  return Array.from(
    {
      length:
        (setting.endTimeMinutes - setting.startTimeMinutes)
        / SLOT_INTERVAL_MINUTES
        + 1,
    },
    (_, index) => {

      const totalMinutes =
        setting.startTimeMinutes
        + index * SLOT_INTERVAL_MINUTES;

      const hours =
        String(
          Math.floor(totalMinutes / 60)
        ).padStart(2, '0');

      const minutes =
        String(
          totalMinutes % 60
        ).padStart(2, '0');

      return {
        id: `${hours}:${minutes}`,
        minutes: totalMinutes,
      };
    }
  );
}


// ========================================
// 日付が正しいか
// ========================================

function isValidDay(day) {
  return day === 'day1' || day === 'day2';
}


// ========================================
// 日本時間を取得
// ========================================

function getJapanDateTime() {
  const formatter =
    new Intl.DateTimeFormat('ja-JP', {
      timeZone: 'Asia/Tokyo',
      year: 'numeric',
      month: '2-digit',
      day: '2-digit',
      hour: '2-digit',
      minute: '2-digit',
      hour12: false,
    });

  const parts =
    formatter.formatToParts(new Date());

  const result = {};

  parts.forEach((part) => {
    if (part.type !== 'literal') {
      result[part.type] = part.value;
    }
  });

  return {
    date:
      `${result.year}-${result.month}-${result.day}`,

    minutes:
      Number(result.hour) * 60
      + Number(result.minute),
  };
}


// ========================================
// 指定日の現在時刻より前の枠を除外
// ========================================
//
// 例：11/21 13:25 の場合
//   13:00 → 除外
//   13:30 → 対象
// 学園祭当日でない日は全枠が対象。
// ========================================

function getActiveSlots(day) {
  const slots = createSlots(day);

  const now = getJapanDateTime();

  if (now.date !== FESTIVAL_DATES[day]) {
    return slots;
  }

  return slots.filter(
    (slot) => slot.minutes >= now.minutes
  );
}


// ========================================
// Firestore初期化
// ========================================

function getFirestoreDb() {
  if (!getApps().length) {

    const privateKey =
      process.env.FIREBASE_PRIVATE_KEY
        ?.replace(/\\n/g, '\n');

    if (
      !process.env.FIREBASE_PROJECT_ID ||
      !process.env.FIREBASE_CLIENT_EMAIL ||
      !privateKey
    ) {
      throw new Error(
        'Firebase Admin environment variables are not configured'
      );
    }

    initializeApp({
      credential: cert({
        projectId:
          process.env.FIREBASE_PROJECT_ID,

        clientEmail:
          process.env.FIREBASE_CLIENT_EMAIL,

        privateKey,
      }),
    });
  }

  return getFirestore();
}


// ========================================
// JSONレスポンス
// ========================================

function json(res, status, body) {
  return res
    .status(status)
    .setHeader(
      'Content-Type',
      'application/json'
    )
    .json(body);
}


// ========================================
// 入力値を整理
// ========================================

function clean(value, maxLength) {
  return typeof value === 'string'
    ? value.trim().slice(0, maxLength)
    : '';
}


function normalizeReservationNumber(value) {
  return clean(value, 20).toUpperCase();
}


// ========================================
// 整理番号作成
// ========================================

function createNumber() {
  return `F-${randomUUID()
    .replace(/-/g, '')
    .slice(0, 8)
    .toUpperCase()}`;
}


// ========================================
// 整理番号順
// ========================================

function sortReservations(first, second) {
  return first.number.localeCompare(
    second.number,
    undefined,
    {
      numeric: true,
      sensitivity: 'base',
    }
  );
}


// ========================================
// スロットのFirestoreドキュメントID
// 例：day1_11:00 / day2_11:30
// ========================================

function createSlotKey(day, slotId) {
  return `${day}_${slotId}`;
}


// ========================================
// 予約の公開用データ
// ========================================

function toPublicReservation(reservation) {
  return {
    number: reservation.number,
    day: reservation.day,
    slotId: reservation.slotId,
    name: reservation.name,
    partySize: reservation.partySize,
    partnerNames: reservation.partnerNames || [],
    verified: reservation.verified === true,
  };
}


// ========================================
// API
// ========================================

export default async function handler(req, res) {

  try {

    const db = getFirestoreDb();


    // ======================================
    // GET
    // ======================================

    if (req.method === 'GET') {

      const requestedDay =
        clean(req.query?.day, 10);

      const day =
        isValidDay(requestedDay)
          ? requestedDay
          : 'day1';


      const slots =
        createSlots(day);


      // --------------------------------------
      // 各枠の予約人数を1回のgetAllで取得
      // --------------------------------------

      const slotRefs =
        slots.map((slot) =>
          db
            .collection('slots')
            .doc(createSlotKey(day, slot.id))
        );

      const slotSnapshots =
        await db.getAll(...slotRefs);

      const counts =
        Object.fromEntries(
          slotSnapshots.map((snapshot, index) => [
            slots[index].id,

            snapshot.exists
              ? snapshot.data().reservedPeople || 0
              : 0,
          ])
        );


      // --------------------------------------
      // スタッフ画面：現在時刻を過ぎていない枠のみ
      // --------------------------------------

      let reservations;

      if (req.query?.view === 'staff') {

        const activeSlots =
          getActiveSlots(day);

        const reservationSnapshots =
          await Promise.all(
            activeSlots.map((slot) =>
              db
                .collection('reservations')
                .where(
                  'slotKey',
                  '==',
                  createSlotKey(day, slot.id)
                )
                .get()
            )
          );

        reservations =
          reservationSnapshots
            .flatMap((snapshot) =>
              snapshot.docs.map((doc) =>
                toPublicReservation(doc.data())
              )
            )
            .sort(sortReservations);
      }


      return json(
        res,
        200,
        {
          counts,

          capacity:
            SLOT_CAPACITY,

          maxPartySize:
            MAX_PARTY_SIZE,

          reservations,
        }
      );
    }


    // ======================================
    // POST以外は拒否
    // ======================================

    if (req.method !== 'POST') {

      return json(
        res,
        405,
        {
          message: 'Method Not Allowed',
        }
      );
    }


    // ======================================
    // 本人確認
    // ======================================

    if (req.body?.action === 'verify') {

      const number =
        normalizeReservationNumber(
          req.body.number
        );

      const name =
        clean(
          req.body.name,
          80
        );


      const reservationSnapshot =
        await db
          .collection('reservations')
          .doc(number)
          .get();


      const reservation =
        reservationSnapshot.exists
          ? reservationSnapshot.data()
          : null;


      if (
        !reservation ||
        reservation.name !== name
      ) {

        return json(
          res,
          404,
          {
            message:
              '整理番号と予約時の名前が一致しません。',
          }
        );
      }


      await reservationSnapshot.ref.update({
        verified: true,
        verifiedAt:
          new Date().toISOString(),
      });


      return json(
        res,
        200,
        {
          reservation: {
            ...toPublicReservation(reservation),
            verified: true,
          },
        }
      );
    }


    // ======================================
    // チェック状態変更
    // ======================================

    if (
      req.body?.action === 'setVerified'
    ) {

      const number =
        normalizeReservationNumber(
          req.body.number
        );

      const verified =
        req.body.verified === true;


      const reservationRef =
        db
          .collection('reservations')
          .doc(number);


      const reservationSnapshot =
        await reservationRef.get();


      if (!reservationSnapshot.exists) {

        return json(
          res,
          404,
          {
            message:
              '整理番号が見つかりません。',
          }
        );
      }


      await reservationRef.update({

        verified,

        verifiedAt:
          verified
            ? new Date().toISOString()
            : null,
      });


      return json(
        res,
        200,
        {
          number,
          verified,
        }
      );
    }


    // ======================================
    // 通常予約（枠は自動割り振り）
    // ======================================

    const day =
      clean(req.body?.day, 10);

    const name =
      clean(req.body?.name, 80);

    const partySize =
      Number(req.body?.partySize);

    const partnerNames =
      Array.isArray(req.body?.partnerNames)
        ? req.body.partnerNames.map(
            (value) => clean(value, 80)
          )
        : [];


    if (!isValidDay(day)) {

      return json(
        res,
        400,
        {
          message:
            '参加日を正しく選択してください。',
        }
      );
    }


    if (
      !Number.isInteger(partySize) ||
      partySize < 1 ||
      partySize > MAX_PARTY_SIZE
    ) {

      return json(
        res,
        400,
        {
          message:
            `参加人数は1〜${MAX_PARTY_SIZE}名で指定してください。`,
        }
      );
    }


    // 同行者は「人数 - 1」名、全員の名前が必要
    if (
      !name ||
      partnerNames.length !== partySize - 1 ||
      partnerNames.some((value) => !value)
    ) {

      return json(
        res,
        400,
        {
          message:
            '代表者名と、人数分の同行者名を入力してください。',
        }
      );
    }


    const number =
      createNumber();

    const createdAt =
      new Date().toISOString();

    const reservationRef =
      db
        .collection('reservations')
        .doc(number);


    // ======================================
    // Transaction：枠の自動割り振り＋予約作成
    // ======================================

    let reservation;

    await db.runTransaction(
      async (transaction) => {

        // リトライ時も毎回現在時刻で再計算
        const activeSlots =
          getActiveSlots(day);

        if (!activeSlots.length) {

          const error =
            new Error(
              '現在、予約できる回がありません。'
            );

          error.code = 'CLOSED';

          throw error;
        }


        const slotRefs =
          activeSlots.map((slot) =>
            db
              .collection('slots')
              .doc(createSlotKey(day, slot.id))
          );

        // 全候補枠を同じTransaction内で読み取る
        const slotSnapshots =
          await transaction.getAll(...slotRefs);


        /*
         * 割り振りルール：
         * 人数が収まる枠のうち、入れた後の余りが最も少ない枠
         * （＝最も20名に近づく枠）を選ぶ。
         * 余りが同じなら、早い時間の枠を優先する。
         */
        let best = null;

        slotSnapshots.forEach((snapshot, index) => {

          const reserved =
            snapshot.exists
              ? snapshot.data().reservedPeople || 0
              : 0;

          const remaining =
            SLOT_CAPACITY - reserved;

          if (remaining < partySize) {
            return;
          }

          const leftover =
            remaining - partySize;

          // 厳密な「<」なので、同率なら先に出てきた早い枠が残る
          if (!best || leftover < best.leftover) {
            best = {
              slot: activeSlots[index],
              ref: slotRefs[index],
              reserved,
              leftover,
            };
          }
        });


        if (!best) {

          const error =
            new Error(
              `${partySize}名で入れる回がありません。人数を減らすか、別の日をお試しください。`
            );

          error.code = 'FULL';

          throw error;
        }


        const slotKey =
          createSlotKey(day, best.slot.id);

        reservation = {
          number,
          day,
          slotId: best.slot.id,
          slotKey,
          name,
          partySize,
          partnerNames,
          verified: false,
          createdAt,
        };


        // 枠の予約人数を加算
        transaction.set(
          best.ref,
          {
            slotKey,
            day,
            slotId: best.slot.id,
            capacity: SLOT_CAPACITY,
            reservedPeople:
              best.reserved + partySize,
          },
          {
            merge: true,
          }
        );


        // 予約を作成
        transaction.create(
          reservationRef,
          reservation
        );
      }
    );


    // ======================================
    // 成功
    // ======================================

    return json(
      res,
      201,
      {
        reservation:
          toPublicReservation(reservation),
      }
    );


  } catch (error) {

    // ======================================
    // 満席・受付終了
    // ======================================

    if (
      error.code === 'FULL' ||
      error.code === 'CLOSED'
    ) {

      return json(
        res,
        409,
        {
          message:
            error.message,
        }
      );
    }


    // ======================================
    // その他エラー
    // ======================================

    console.error(error);

    return json(
      res,
      500,
      {
        message:
          '予約システムに接続できません。管理者にご確認ください。',
      }
    );
  }
}

//  /reservation.js
// ========================================
// 学園祭の日付・時間設定
// ========================================

const FESTIVAL_DATES = {
    day1: '2026-11-21',
    day2: '2026-11-22',
};

const DAY_LABELS = {
    day1: '1日目（11月21日）',
    day2: '2日目（11月22日）',
};

// 両日とも 11:00〜16:00
const DAY_SETTINGS = {
    day1: {
        startTimeMinutes: 11 * 60,
        endTimeMinutes: 16 * 60,
    },

    day2: {
        startTimeMinutes: 11 * 60,
        endTimeMinutes: 16 * 60,
    },
};

const SLOT_INTERVAL_MINUTES = 30;

// 1枠の定員（人数）
const SLOT_CAPACITY = 20;

// 1回の予約の最大人数
const MAX_PARTY_SIZE = 4;


// ========================================
// DOM
// ========================================

const festivalDay = document.getElementById('festivalDay');
const partySizeSelect = document.getElementById('partySize');
const companionFields = document.getElementById('companionFields');
const availability = document.getElementById('availability');
const reservationForm = document.getElementById('reservationForm');
const formFeedback = document.getElementById('formFeedback');

// 現在表示している日の各枠の予約人数
let currentCounts = {};


// ========================================
// 選択された日の時間枠を作成
// ========================================

function createSlots(day) {
    const setting = DAY_SETTINGS[day];

    if (!setting) {
        return [];
    }

    return Array.from(
        {
            length:
                (setting.endTimeMinutes - setting.startTimeMinutes)
                / SLOT_INTERVAL_MINUTES
                + 1
        },
        (_, index) => {
            const totalMinutes =
                setting.startTimeMinutes
                + index * SLOT_INTERVAL_MINUTES;

            const hours = String(
                Math.floor(totalMinutes / 60)
            ).padStart(2, '0');

            const minutes = String(
                totalMinutes % 60
            ).padStart(2, '0');

            return {
                id: `${hours}:${minutes}`,
                minutes: totalMinutes,
            };
        }
    );
}


// ========================================
// フィードバック表示
// ========================================

function showFeedback(element, message, isError = true) {
    element.textContent = message;
    element.className = `feedback ${isError ? 'error' : 'success'}`;
}


// ========================================
// 日本時間を取得
// ========================================

function getJapanDateTime() {
    const formatter = new Intl.DateTimeFormat('ja-JP', {
        timeZone: 'Asia/Tokyo',
        year: 'numeric',
        month: '2-digit',
        day: '2-digit',
        hour: '2-digit',
        minute: '2-digit',
        hour12: false,
    });

    const parts = formatter.formatToParts(new Date());

    const result = {};

    parts.forEach((part) => {
        if (part.type !== 'literal') {
            result[part.type] = part.value;
        }
    });

    return {
        date: `${result.year}-${result.month}-${result.day}`,
        minutes:
            Number(result.hour) * 60
            + Number(result.minute),
    };
}


// ========================================
// 開始時刻を過ぎた枠か
// ========================================

function isSlotPast(day, slotMinutes) {
    const now = getJapanDateTime();

    if (now.date !== FESTIVAL_DATES[day]) {
        return false;
    }

    return slotMinutes < now.minutes;
}


// ========================================
// 同行者の入力欄を人数に合わせて作成
// ========================================

function renderCompanionInputs() {
    // 入力済みの内容は可能な限り残す
    const previousValues =
        [...companionFields.querySelectorAll('input')]
            .map((input) => input.value);

    const partySize = Number(partySizeSelect.value);

    companionFields.replaceChildren();

    for (let index = 0; index < partySize - 1; index += 1) {
        const label = document.createElement('label');
        label.textContent = `同行者${index + 1}名`;

        const input = document.createElement('input');
        input.type = 'text';
        input.maxLength = 80;
        input.required = true;
        input.className = 'companion-input';
        input.value = previousValues[index] || '';

        label.appendChild(input);
        companionFields.appendChild(label);
    }

    companionFields.hidden = partySize <= 1;
}


// ========================================
// 空き状況を画面に表示
//
// 枠は自動で割り振られるため、
// 「いま入れる最大人数」だけを案内します。
// ========================================

function renderAvailability() {
    const day = festivalDay.value;

    const remainingList =
        createSlots(day)
            .filter((slot) => !isSlotPast(day, slot.minutes))
            .map((slot) =>
                SLOT_CAPACITY - (currentCounts[slot.id] || 0)
            );

    const maxRemaining =
        remainingList.length
            ? Math.max(0, ...remainingList)
            : 0;

    const totalRemaining =
        remainingList.reduce((sum, value) => sum + value, 0);

    const submitButton =
        reservationForm.querySelector('button[type="submit"]');

    // 入れない人数は選べないようにする
    [...partySizeSelect.options].forEach((option) => {
        option.disabled = Number(option.value) > maxRemaining;
    });

    // 現在の選択が選べなくなった場合、選べる最大人数に合わせる
    if (maxRemaining > 0 && Number(partySizeSelect.value) > maxRemaining) {
        partySizeSelect.value = String(Math.min(MAX_PARTY_SIZE, maxRemaining));
        renderCompanionInputs();
    }

    if (maxRemaining === 0) {
        submitButton.disabled = true;

        availability.textContent = '';

        const now = getJapanDateTime();

        showFeedback(
            formFeedback,
            now.date === FESTIVAL_DATES[day]
                ? '現在、予約できる回がありません。'
                : 'この日の予約は満員です。',
            true
        );
    } else {
        submitButton.disabled = false;

        availability.textContent =
            `残り ${totalRemaining} 名分（1組で最大 ${Math.min(MAX_PARTY_SIZE, maxRemaining)} 名まで予約できます）`;
    }
}


// ========================================
// 選択日の空き状況を取得
// ========================================

async function loadSlots() {
    try {
        const day = festivalDay.value;

        const response = await fetch(
            `/api/reservations?day=${encodeURIComponent(day)}`
        );

        const data = await response.json();

        if (!response.ok) {
            throw new Error(data.message);
        }

        currentCounts = data.counts || {};

        renderAvailability();
    } catch (error) {
        currentCounts = {};

        renderAvailability();

        showFeedback(
            formFeedback,
            error.message || '空き状況を取得できませんでした。',
            true
        );
    }
}


// ========================================
// 日付・人数変更時
// ========================================

festivalDay.addEventListener('change', async () => {
    showFeedback(formFeedback, '', false);

    await loadSlots();
});

partySizeSelect.addEventListener('change', renderCompanionInputs);


// ========================================
// 予約送信
// ========================================

reservationForm.addEventListener('submit', async (event) => {
    event.preventDefault();

    const button =
        reservationForm.querySelector('button[type="submit"]');

    button.disabled = true;

    showFeedback(
        formFeedback,
        '予約を確認しています…',
        false
    );

    try {
        const selectedDay = festivalDay.value;
        const partySize = Number(partySizeSelect.value);
        const name = document.getElementById('name').value;

        const partnerNames =
            [...companionFields.querySelectorAll('input')]
                .map((input) => input.value);

        const response = await fetch('/api/reservations', {
            method: 'POST',

            headers: {
                'Content-Type': 'application/json',
            },

            body: JSON.stringify({
                day: selectedDay,
                partySize,
                name,
                partnerNames,
            }),
        });

        const data = await response.json();

        if (!response.ok) {
            throw new Error(data.message);
        }

        const reservation = data.reservation;

        // 結果を表示
        document.getElementById('reservationNumber').textContent =
            reservation.number;

        document.getElementById('reservationDay').textContent =
            DAY_LABELS[reservation.day] || reservation.day;

        document.getElementById('reservationSlot').textContent =
            `${reservation.slotId}〜`;

        document.getElementById('reservationPartySize').textContent =
            `${reservation.partySize}名`;

        document.getElementById('reservationName').textContent =
            reservation.name;

        document.getElementById('reservationPartnerNames').textContent =
            reservation.partnerNames.length
                ? reservation.partnerNames.join('、')
                : 'なし';

        document.getElementById('numberPanel').hidden = false;

        // フォームをリセット（参加日は維持）
        reservationForm.reset();
        festivalDay.value = selectedDay;
        renderCompanionInputs();

function getJapanDateTime() {
    const formatter = new Intl.DateTimeFormat('ja-JP', {
        timeZone: 'Asia/Tokyo',
        year: 'numeric',
        month: '2-digit',
        day: '2-digit',
        hour: '2-digit',
        minute: '2-digit',
        hour12: false,
    });

    const parts = formatter.formatToParts(new Date());

    const result = {};

    parts.forEach((part) => {
        if (part.type !== 'literal') {
            result[part.type] = part.value;
        }
    });

    return {
        date: `${result.year}-${result.month}-${result.day}`,

        minutes:
            Number(result.hour) * 60
            + Number(result.minute),
    };
}

import { cert, getApps, initializeApp } from 'firebase-admin/app';
import { getFirestore } from 'firebase-admin/firestore';

function getFirestoreDb() {
  if (!getApps().length) {
    const privateKey = process.env.FIREBASE_PRIVATE_KEY?.replace(/\\n/g, '\n');

    if (
      !process.env.FIREBASE_PROJECT_ID ||
      !process.env.FIREBASE_CLIENT_EMAIL ||
      !privateKey
    ) {
      throw new Error('Firebase Admin environment variables are not configured');
    }

    initializeApp({
      credential: cert({
        projectId: process.env.FIREBASE_PROJECT_ID,
        clientEmail: process.env.FIREBASE_CLIENT_EMAIL,
        privateKey,
      }),
    });
  }

  return getFirestore();
}

function json(res, status, body) {
  return res
    .status(status)
    .setHeader('Content-Type', 'application/json')
    .json(body);
}

function invalid(message) {
  const error = new Error(message);
  error.status = 400;
  return error;
}

function cleanString(value, maxLength) {
  return typeof value === 'string' ? value.trim().slice(0, maxLength) : '';
}

function documentId(value) {
  const id = cleanString(value, 128);
  if (!/^[A-Za-z0-9_-]+$/.test(id)) {
    throw invalid('データのIDが正しくありません。');
  }
  return id;
}

function normalizeProduct(value) {
  const price = Number(value?.price);
  const name = cleanString(value?.name, 120);
  if (!name || !Number.isSafeInteger(price) || price < 0) {
    throw invalid('商品名または価格が正しくありません。');
  }
  return { id: documentId(value?.id), name, price };
}

function normalizeReceipt(value) {
  const id = documentId(value?.id);
  const date = cleanString(value?.date, 10);
  const time = cleanString(value?.time, 8);
  if (!/^\d{4}-\d{2}-\d{2}$/.test(date) || !/^\d{2}:\d{2}(:\d{2})?$/.test(time)) {
    throw invalid('会計日時が正しくありません。');
  }
  if (!Array.isArray(value?.items) || value.items.length > 100) {
    throw invalid('会計明細が正しくありません。');
  }

  const items = value.items.map((item) => {
    const name = cleanString(item?.name, 120);
    const price = Number(item?.price);
    const qty = Number(item?.qty);
    if (!name || !Number.isSafeInteger(price) || price < 0 || !Number.isSafeInteger(qty) || qty < 1) {
      throw invalid('会計明細が正しくありません。');
    }
    return { name, price, qty };
  });

  const receipt = { id, date, time, items };
  for (const field of ['total', 'paid', 'change']) {
    if (value[field] == null) {
      receipt[field] = null;
    } else if (Number.isSafeInteger(Number(value[field])) && Number(value[field]) >= 0) {
      receipt[field] = Number(value[field]);
    } else {
      throw invalid('会計金額が正しくありません。');
    }
  }
  return receipt;
}

function normalizeClosing(value) {
  const date = cleanString(value?.date, 10);
  if (!/^\d{4}-\d{2}-\d{2}$/.test(date)) {
    throw invalid('レジ締めの日付が正しくありません。');
  }
  const amount = (input) => {
    const text = input == null ? '' : String(input);
    if (!/^\d{0,12}$/.test(text)) {
      throw invalid('レジ締め金額が正しくありません。');
    }
    return text;
  };
  return { date, float: amount(value.float), actual: amount(value.actual) };
}

async function writeInChunks(db, operations) {
  for (let start = 0; start < operations.length; start += 400) {
    const batch = db.batch();
    operations.slice(start, start + 400).forEach((operation) => operation(batch));
    await batch.commit();
  }
}

async function collectionData(collection) {
  const snapshot = await collection.get();
  return snapshot.docs.map((doc) => ({ id: doc.id, ...doc.data() }));
}

async function handler(req, res) {
  try {
    const store = cleanString(req.method === 'GET' ? req.query?.store : req.body?.store, 20);
    if (store !== 'asumin' && store !== 'yuna') {
      return json(res, 400, { message: '店舗を正しく指定してください。' });
    }

    const db = getFirestoreDb();
    const storeRef = db.collection('registers').doc(store);
    const collections = {
      products: storeRef.collection('products'),
      receipts: storeRef.collection('receipts'),
      closing: storeRef.collection('closing'),
    };

    if (req.method === 'GET') {
      const [products, receipts, closing] = await Promise.all([
        collectionData(collections.products),
        collectionData(collections.receipts),
        collectionData(collections.closing),
      ]);
      return json(res, 200, { products, receipts, closing });
    }

    if (req.method !== 'POST') {
      return json(res, 405, { message: 'Method Not Allowed' });
    }

    const body = req.body || {};
    switch (body.action) {
      case 'product-save': {
        const product = normalizeProduct(body.product);
        await collections.products.doc(product.id).set(product);
        break;
      }
      case 'product-delete':
        await collections.products.doc(documentId(body.id)).delete();
        break;
      case 'receipt-save': {
        const receipt = normalizeReceipt(body.receipt);
        await collections.receipts.doc(receipt.id).set(receipt);
        break;
      }
      case 'receipt-delete':
        await collections.receipts.doc(documentId(body.id)).delete();
        break;
      case 'closing-save': {
        const closing = normalizeClosing(body.closing);
        await collections.closing.doc(closing.date).set(closing);
        break;
      }
      case 'receipts-replace': {
        if (!Array.isArray(body.receipts)) {
          throw invalid('会計データの形式が正しくありません。');
        }
        const receipts = body.receipts.map(normalizeReceipt);
        const receiptIds = new Set(receipts.map((receipt) => receipt.id));
        if (receiptIds.size !== receipts.length) {
          throw invalid('会計データに重複したIDがあります。');
        }
        const existing = await collections.receipts.get();
        const operations = [
          ...existing.docs
            .filter((doc) => !receiptIds.has(doc.id))
            .map((doc) => (batch) => batch.delete(doc.ref)),
          ...receipts.map((receipt) => (batch) =>
            batch.set(collections.receipts.doc(receipt.id), receipt)),
        ];
        await writeInChunks(db, operations);
        break;
      }
      case 'wipe': {
        const snapshots = await Promise.all(
          Object.values(collections).map((collection) => collection.get())
        );
        const operations = snapshots.flatMap((snapshot) =>
          snapshot.docs.map((doc) => (batch) => batch.delete(doc.ref))
        );
        await writeInChunks(db, operations);
        break;
      }
      case 'migrate': {
        const [productsSnapshot, receiptsSnapshot, closingSnapshot] = await Promise.all([
          collections.products.get(),
          collections.receipts.get(),
          collections.closing.get(),
        ]);
        if (
          productsSnapshot.size ||
          receiptsSnapshot.size ||
          closingSnapshot.size
        ) {
          return json(res, 200, { migrated: false });
        }

        const products = Array.isArray(body.products)
          ? body.products.map(normalizeProduct)
          : [];
        const receipts = Array.isArray(body.receipts)
          ? body.receipts.map(normalizeReceipt)
          : [];
        const closing = body.closing && typeof body.closing === 'object'
          ? Object.entries(body.closing).map(([date, data]) =>
            normalizeClosing({ ...data, date }))
          : [];
        const operations = [
          ...products.map((product) => (batch) =>
            batch.set(collections.products.doc(product.id), product)),
          ...receipts.map((receipt) => (batch) =>
            batch.set(collections.receipts.doc(receipt.id), receipt)),
          ...closing.map((entry) => (batch) =>
            batch.set(collections.closing.doc(entry.date), entry)),
        ];
        await writeInChunks(db, operations);
        return json(res, 200, { migrated: true });
      }
      default:
        return json(res, 400, { message: '操作を正しく指定してください。' });
    }

    return json(res, 200, { ok: true });
  } catch (error) {
    if (error.status === 400) {
      return json(res, 400, { message: error.message });
    }
    console.error(error);
    return json(res, 500, { message: 'Firestoreに保存できませんでした。' });
  }
}

export default handler;

/* かんたんレジ - Firestore同期版
   Firestoreを主保存先とし、localStorageは一時キャッシュと旧データ移行に使用
   ※ iPhoneでは「ホーム画面に追加」して使うと自動削除(ITP)の対象外になります */
(() => {
  'use strict';

  /* ---------- 保存層 ---------- */
  const STORE = window.registerCloud.store;
  const KEY_PREFIX = 'kr.' + STORE + '.v2.';
  const KEY = {
    products: KEY_PREFIX + 'products',
    receipts: KEY_PREFIX + 'receipts',
    closing: KEY_PREFIX + 'closing',
    cart: KEY_PREFIX + 'cart'
  };
  const mem = {}; // localStorage が使えないときの退避（リロードで消える）
  const cloud = window.registerCloud;

  async function cloudWrite(action, data) {
    try {
      await cloud.write(action, data);
      return true;
    } catch (e) {
      toast('⚠ Firestoreへの保存に失敗しました: ' + e.message);
      return false;
    }
  }

  function showCloudError(error) {
    toast('⚠ Firestoreからデータを取得できません: ' + error.message);
  }

  function load(key, fallback) {
    try {
      const s = localStorage.getItem(key);
      return s ? JSON.parse(s) : fallback;
    } catch (e) {
      return key in mem ? mem[key] : fallback;
    }
  }
  function save(key, val) {
    mem[key] = val;
    try {
      localStorage.setItem(key, JSON.stringify(val));
    } catch (e) {
      toast('⚠ 保存できません（プライベートモード/容量不足の可能性）');
    }
  }

  // 消されにくくするようブラウザに要求（許可はブラウザ次第）
  try {
    if (navigator.storage && navigator.storage.persist) navigator.storage.persist();
  } catch (e) { /* 無視 */ }

  function parseRaw(text) {
    const map = new Map();
    const lines = text.split(/\r?\n/).filter(l => l.trim());
    for (let n = 0; n < lines.length; n++) {
      const c = lines[n].split('\t');
      if (c.length < 6 || !/^\d{4}-\d{2}-\d{2}$/.test(c[0]) || isNaN(+c[4]) || isNaN(+c[5])) {
        throw new Error((n + 1) + '行目の形式が正しくありません');
      }
      const id = c[2];
      if (!map.has(id)) map.set(id, { id, date: c[0], time: c[1], items: [] });
      map.get(id).items.push({ name: c[3], price: +c[4], qty: +c[5] });
    }
    return [...map.values()];
  }

    }

    Object.values(KEY).forEach(k => {
      delete mem[k];
      try { localStorage.removeItem(k); } catch (e) { /* 無視 */ }
    });
    products = [];
    receipts = [];
    closing = {};
    cart = [];
    editMode = false;
    $('paid').value = '';
    renderTiles();
    renderCart();
    renderSales();
    toast('全データを削除しました');
  });

  /* ---------- 起動 ---------- */
  async function initialize() {
    document.body.inert = true;
    try {
      let data = await cloud.load();
      if (!data.products.length && !data.receipts.length && !data.closing.length) {
        if (products.length || receipts.length || Object.keys(closing).length) {
          await cloud.write('migrate', { products, receipts, closing });
          data = await cloud.load();
        }
      }
      products = data.products;
      receipts = data.receipts;
      closing = Object.fromEntries(data.closing.map(({ date, ...value }) => [date, value]));
      save(KEY.products, products);
      save(KEY.receipts, receipts);
      save(KEY.closing, closing);
    } catch (error) {
      showCloudError(error);
    }
    finally {
      selDate = dstr(new Date());
      renderTiles();
      renderCart();
      document.body.inert = false;
    }
  }
  initialize();
})();

  /* ---------- カート ---------- */
  function addToCart(p) {
    const ex = cart.find(c => c.id === p.id);
    if (ex) ex.qty++;
    else cart.push({ id: p.id, name: p.name, price: p.price, qty: 1 });

  function renderCart() {

    box.replaceChildren();
    if (!cart.length) box.append(h('p', { class: 'muted' }, '商品をタップして追加'));

  function renderChange() {

    save(KEY.cart, cart);

    renderCart();

  $('payBtn').addEventListener('click', async () => {
    if (!cart.length) return toast('商品が選ばれていません');

    $('paid').value = '';
    renderCart();
    if (saved) {
      toast(change === null ? '会計しました ' + yen(total) : '会計しました　おつり ' + yen(change));
    }
  });

  /* ---------- 売上 ---------- */
  const dayReceipts = () => receipts.filter(r => r.date === selDate)
    .sort((a, b) => a.time.localeCompare(b.time));

  function renderSales() {
    if (!selDate) selDate = dstr(new Date());
    $('date').value = selDate;

    // 日付チップ（売上のある日）
    const dates = [...new Set(receipts.map(r => r.date))].sort().reverse().slice(0, 14);
    const chips = $('chips');
    chips.replaceChildren();
    dates.forEach(d => {
      const sum = receipts.filter(r => r.date === d).reduce((s, r) => s + receiptTotal(r), 0);
      chips.append(h('button', {
        class: 'chip' + (d === selDate ? ' on' : ''), type: 'button',
        onclick: () => { selDate = d; renderSales(); }
      }, d.slice(5).replace('-', '/') + ' ' + yen(sum)));
    });

    const list = dayReceipts();
    const total = list.reduce((s, r) => s + receiptTotal(r), 0);
    const count = list.reduce((s, r) => s + r.items.reduce((a, i) => a + i.qty, 0), 0);

    // 統計
    const stat = (label, val) => h('div', { class: 'stat' }, h('small', null, label), h('b', null, val));
    $('stats').replaceChildren(
      stat('売上合計', yen(total)),
      stat('会計数', list.length + '件'),
      stat('販売点数', count + '点'),
      stat('客単価', list.length ? yen(Math.round(total / list.length)) : '—')
    );

    // レジ締め
    const cl = closing[selDate] || {};
    $('cFloat').value = cl.float ?? '';
    $('cActual').value = cl.actual ?? '';
    renderClosing();

    // 商品別
    const agg = {};
    list.forEach(r => r.items.forEach(i => {
      const k = i.name + '\t' + i.price;
      agg[k] = agg[k] || { name: i.name, price: i.price, qty: 0 };
      agg[k].qty += i.qty;
    }));
    const rows = Object.values(agg).sort((a, b) => b.price * b.qty - a.price * a.qty);
    $('byItem').replaceChildren(rows.length
      ? h('table', null,
        h('thead', null, h('tr', null, ...['商品', '単価', '数量', '小計'].map(t => h('th', null, t)))),

        }, '削除')));
    });

    $('raw').value = buildRaw();
  }

  /* ---------- 保存テキスト（バックアップ/復元） ---------- */
  function buildRaw() {
    const lines = [];
    receipts.slice()
      .sort((a, b) => (a.date + a.time).localeCompare(b.date + b.time))
      .forEach(r => r.items.forEach(i =>
        lines.push([r.date, r.time, r.id, i.name, i.price, i.qty, i.price * i.qty].join('\t'))));
    return lines.join('\n');
  }

  function parseRaw(text) {
    const map = new Map();
    const lines = text.split(/\r?\n/).filter(l => l.trim());
    for (let n = 0; n < lines.length; n++) {
      const c = lines[n].split('\t');
      if (c.length < 6 || !/^\d{4}-\d{2}-\d{2}$/.test(c[0]) || isNaN(+c[4]) || isNaN(+c[5])) {
        throw new Error((n + 1) + '行目の形式が正しくありません');
      }
      const id = c[2];
      if (!map.has(id)) map.set(id, { id, date: c[0], time: c[1], items: [] });
      map.get(id).items.push({ name: c[3], price: +c[4], qty: +c[5] });
    }
    return [...map.values()];
  }

  $('copyBtn').addEventListener('click', async () => {
    const t = $('raw').value;
    try {
      await navigator.clipboard.writeText(t);
      toast('コピーしました');
    } catch (e) {
      $('raw').select();
      document.execCommand && document.execCommand('copy');
      toast('コピーしました');
    }
  });

  /* ---------- ファイルに保存（iPhone: 共有シート →「"ファイル"に保存」） ---------- */
  $('fileBtn').addEventListener('click', async () => {
    const text = buildRaw();
    if (!text) return toast('保存するデータがありません');
    const now = new Date();
    const name = 'register_' + dstr(now) + '_' + p2(now.getHours()) + p2(now.getMinutes()) + '.txt';

    // 1) 共有シート（iOS15+ / HTTPS が必要）
    try {
      const file = new File([text], name, { type: 'text/plain' });
      if (navigator.canShare && navigator.canShare({ files: [file] })) {
        await navigator.share({ files: [file], title: name });
        return toast('書き出しました');
      }
    } catch (e) {
      if (e && e.name === 'AbortError') return; // 共有シートを閉じただけ
      /* それ以外の失敗はダウンロードにフォールバック */
    }

    // 2) ダウンロード（ファイルアプリの「ダウンロード」に入る）
    const url = URL.createObjectURL(new Blob([text], { type: 'text/plain' }));
    const a = h('a', { href: url, download: name });
    document.body.append(a);
    a.click();
    a.remove();
    setTimeout(() => URL.revokeObjectURL(url), 10000);
    toast('ダウンロードしました');
  });

  $('saveBtn').addEventListener('click', async () => {
    let parsed;
    try {
      parsed = parseRaw($('raw').value);
    } catch (e) {
      return toast(e.message);
    }
    if (!confirm('全期間の売上を、このテキストの内容（' + parsed.length + '件の会計）で置き換えます。よろしいですか？')) return;
    receipts = parsed;
    save(KEY.receipts, receipts);
    const saved = await cloudWrite('receipts-replace', { receipts });
    renderSales();
    if (saved) toast('保存しました');
  });

  /* ---------- 全データ削除 ---------- */
  $('wipeBtn').addEventListener('click', async () => {
    if (!confirm('保存されている全データ（商品・売上・レジ締め）を完全に削除します。\n元に戻せません。\n\n事前に「保存テキスト」をコピーして控えましたか？\n続行しますか？')) return;
    const v = prompt('本当に削除する場合は学籍番号を入力してください');
    // 大文字小文字・全角半角の違いは無視して比較（両辺を同じ形に揃える）
    const fold = s => s.normalize('NFKC').trim().toLowerCase();
    if (v === null || fold(v) !== fold('2502CO0011')) return toast('キャンセルしました');
    if (!await cloudWrite('wipe')) return;
    // このアプリのキー(kr.*)だけを消す。同じオリジンの他のデータには触れない
    Object.values(KEY).forEach(k => {
      delete mem[k];
      try { localStorage.removeItem(k); } catch (e) { /* 無視 */ }
    });
    products = [];
    receipts = [];
    closing = {};
    cart = [];
    editMode = false;
    $('paid').value = '';
    renderTiles();
    renderCart();
    renderSales();
    toast('全データを削除しました');
  });

  /* ---------- 起動 ---------- */
  async function initialize() {
    document.body.inert = true;
    try {
      let data = await cloud.load();
      if (!data.products.length && !data.receipts.length && !data.closing.length) {
        if (products.length || receipts.length || Object.keys(closing).length) {
          await cloud.write('migrate', { products, receipts, closing });
          data = await cloud.load();
        }
      }
      products = data.products;
      receipts = data.receipts;
      closing = Object.fromEntries(data.closing.map(({ date, ...value }) => [date, value]));
      save(KEY.products, products);
      save(KEY.receipts, receipts);
      save(KEY.closing, closing);
    } catch (error) {
      showCloudError(error);
    }
    finally {
      selDate = dstr(new Date());
      renderTiles();
      renderCart();
      document.body.inert = false;
    }
  }
  initialize();
})();
